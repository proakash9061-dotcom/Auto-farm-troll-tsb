-- ============ HELPERS ============
local function isAlive(p)
    local c = p.Character
    if not c then return false end
    local h = c:FindFirstChildOfClass("Humanoid")
    return h and h.Health > 0
end

local function getRoot(p)
    local c = p.Character
    return c and c:FindFirstChild("HumanoidRootPart")
end

local function getTargetPart(p, mode)
    local c = p.Character
    if not c then return nil end
    if mode == "Head" then
        return c:FindFirstChild("Head")
    end
    return c:FindFirstChild("UpperTorso")
        or c:FindFirstChild("Torso")
        or c:FindFirstChild("HumanoidRootPart")
end

local function isTeammate(p)
    return p.Team ~= nil and p.Team == LP.Team
end

local function isVisible(part, ignore)
    if not part then return false end
    local origin = Camera.CFrame.Position
    local dir = part.Position - origin
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = ignore or { LP.Character, Camera }
    params.IgnoreWater = true
    local hit = Workspace:Raycast(origin, dir, params)
    return hit == nil or hit.Instance:IsDescendantOf(part.Parent)
end

-- ============ TARGETING ============
local function getClosest(fov, teamCheck, mode, wallCheck)
    local closest, shortest = nil, fov
    local cx, cy = Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2

    for _, p in ipairs(Players:GetPlayers()) do
        if p == LP or not isAlive(p) then continue end
        if teamCheck and isTeammate(p) then continue end

        local part = getTargetPart(p, mode)
        if not part then continue end

        local sp, onScreen = Camera:WorldToViewportPoint(part.Position)
        if not onScreen then continue end

        local d = (Vector2.new(sp.X, sp.Y) - Vector2.new(cx, cy)).Magnitude
        if d < shortest then
            if wallCheck and not isVisible(part, { LP.Character, Camera }) then continue end
            shortest = d
            closest = p
        end
    end
    return closest
end

-- ============ SILENT AIM (reentrancy-guarded) ============
local hookBusy = false
local mt = getrawmetatable(game)
local oldNamecall = mt.__namecall
setreadonly(mt, false)

mt.__namecall = newcclosure(function(self, ...)
    local method = getnamecallmethod()

    if hookBusy
       or not Config.SilentAim.Enabled
       or checkcaller()
       or typeof(method) ~= "string" then
        return oldNamecall(self, ...)
    end

    if method ~= "FindPartOnRay"
       and method ~= "FindPartOnRayWithIgnoreList"
       and method ~= "FindPartOnRayWithWhitelist"
       and method ~= "Raycast" then
        return oldNamecall(self, ...)
    end

    if Config.SilentAim.HitChance < 100
       and math.random(1, 100) > Config.SilentAim.HitChance then
        return oldNamecall(self, ...)
    end

    hookBusy = true
    local ok, ret = pcall(function()
        local t = getClosest(
            Config.SilentAim.FOV, Config.SilentAim.TeamCheck,
            Config.SilentAim.Mode, Config.SilentAim.WallCheck
        )
        if not t then return oldNamecall(self, ...) end

        local part = getTargetPart(t, Config.SilentAim.Mode)
        local root = getRoot(t)
        if not part then return oldNamecall(self, ...) end

        local aimPos = part.Position
        if root and Config.SilentAim.Prediction > 0 then
            aimPos = aimPos + root.AssemblyLinearVelocity * Config.SilentAim.Prediction
        end

        local args = { ... }

        if method:sub(1, 13) == "FindPartOnRay" then
            local oldRay = args[1]
            if typeof(oldRay) == "Ray" then
                args[1] = Ray.new(oldRay.Origin, aimPos - oldRay.Origin)
            end
        elseif method == "Raycast"
           and typeof(self) == "Instance"
           and self:IsA("Workspace") then
            local origin = args[1]
            local dir = args[2]
            if typeof(origin) == "Vector3" then
                local nd = aimPos - origin
                if typeof(dir) == "Vector3" and dir.Magnitude > 0 then
                    nd = nd.Unit * dir.Magnitude
                end
                args[2] = nd
            end
        end

        return oldNamecall(self, unpack(args))
    end)

    hookBusy = false
    if ok then return ret end
    return oldNamecall(self, ...)
end)
setreadonly(mt, true)

-- ============ AUTO FARM — mount on back + headshot ============
-- Grabs a live target, flies LP.Character behind them (mount), fires on head,
-- then re-targets the next player. Full loop, camera tracks current target.
local farmTarget = nil
local farmState = "idle"    -- idle | approach | mount | attack | retarget
local farmTimer = 0
local farmShots = 0

local function pickFarmTarget()
    local best, bestDist = nil, math.huge
    local myRoot = getRoot(LP)
    if not myRoot then return nil end

    for _, p in ipairs(Players:GetPlayers()) do
        if p == LP or not isAlive(p) then continue end
        if Config.SilentAim.TeamCheck and isTeammate(p) then continue end
        local root = getRoot(p)
        if not root then continue end
        local d = (root.Position - myRoot.Position).Magnitude
        if d < bestDist then
            best, bestDist = p, d
        end
    end
    return best
end

local function applyFlyMove(myRoot, destCFrame, speed, dt)
    local delta = destCFrame.Position - myRoot.Position
    local dist = delta.Magnitude
    if dist < 0.1 then return true end
    local step = math.min(dist, speed * dt)
    local dir = delta.Unit
    myRoot.CFrame = CFrame.new(myRoot.Position + dir * step, destCFrame.Position)
    return false
end

RunService.Heartbeat:Connect(function(dt)
    if not Config.AutoFarm.Enabled then
        farmState = "idle"
        farmTarget = nil
        return
    end

    local myChar = LP.Character
    local myRoot = getRoot(LP)
    if not myChar or not myRoot then return end

    -- Re-pick target if dead or lost
    if not farmTarget or not isAlive(farmTarget) then
        farmTarget = pickFarmTarget()
        farmState = "approach"
        farmShots = 0
    end
    if not farmTarget then return end

    local tRoot = getRoot(farmTarget)
    local tHead = getTargetPart(farmTarget, Config.AutoFarm.Mode)
    if not tRoot or not tHead then
        farmTarget = nil
        return
    end

    -- Camera tracking
    if Config.AutoFarm.CameraTrack then
        local look = CFrame.new(Camera.CFrame.Position, tHead.Position)
        Camera.CFrame = Camera.CFrame:Lerp(look, 0.25)
    end

    -- Destination: behind + above target
    local behind = tRoot.CFrame.LookVector * -Config.AutoFarm.OffsetZ
    local dest = CFrame.new(tRoot.Position + behind + Vector3.new(0, Config.AutoFarm.OffsetY, 0))

    if farmState == "approach" or farmState == "retarget" then
        local arrived = applyFlyMove(myRoot, dest, Config.AutoFarm.Speed, dt)
        if arrived then
            farmState = "mount"
            farmTimer = 0
        end
    elseif farmState == "mount" then
        -- Lock position to back of target
        myRoot.CFrame = CFrame.new(dest.Position, tRoot.Position)
        farmTimer = farmTimer + dt
        if farmTimer >= Config.AutoFarm.KillDelay then
            farmState = "attack"
        end
    elseif farmState == "attack" then
        -- Fire on head via virtual mouse
        local headPos = tHead.Position + tRoot.AssemblyLinearVelocity * 0.1
        local ray = Ray.new(Camera.CFrame.Position, headPos - Camera.CFrame.Position)
        local hit, pos = Workspace:FindPartOnRayWithIgnoreList(ray, { myChar, Camera })
        local virtualMouse = LP:GetMouse()
        if virtualMouse then
            virtualMouse.Hit = CFrame.new(pos or headPos)
        end
        pcall(function() LP:GetMouse():Click() end)

        farmShots = farmShots + 1
        if farmShots >= Config.AutoFarm.HitsPerTarget then
            farmShots = 0
            farmState = "retarget"
            farmTarget = nil
        end
    end
end)

local flyVel = Vector3.zero
RunService.RenderStepped:Connect(function(dt)
    if not Config.Fly.Enabled then return end
    local char = LP.Character
    local root = getRoot(LP)
    if not root then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if hum then hum.PlatformStand = true end

    local dir = Vector3.zero
    local camCF = Camera.CFrame
    if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir += camCF.LookVector end
    if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir -= camCF.LookVector end
    if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir -= camCF.RightVector end
    if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir += camCF.RightVector end
    if UserInputService:IsKeyDown(Enum.KeyCode.Space) then dir += Vector3.new(0, 1, 0) end
    if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then dir -= Vector3.new(0, 1, 0) end

    if dir.Magnitude > 0 then
        root.CFrame = root.CFrame + dir.Unit * Config.Fly.Speed * dt
    end
end)

-- ============ ESP ============
local espCache = {}

local function makeESP(p)
    if espCache[p] then return end
    local bb = Instance.new("BillboardGui")
    bb.Name = "PlayerTag"
    bb.Size = UDim2.new(0, 100, 0, 60)
    bb.AlwaysOnTop = true
    bb.ResetOnSpawn = false
    bb.StudsOffset = Vector3.new(0, 3, 0)

    local box = Instance.new("Frame", bb)
    box.BackgroundTransparency = 1
    box.Size = UDim2.new(1, 0, 1, 0)
    local stroke = Instance.new("UIStroke", box)
    stroke.Thickness = 1.5
    stroke.Color = Config.ESP.Color

    local name = Instance.new("TextLabel", bb)
    name.BackgroundTransparency = 1
    name.Size = UDim2.new(1, 0, 0, 14)
    name.Position = UDim2.new(0, 0, 0, -16)
    name.TextColor3 = Config.ESP.Color
    name.TextStrokeTransparency = 0
    name.TextSize = 12
    name.Font = Enum.Font.GothamBold
    name.Text = p.Name

    local hpBg = Instance.new("Frame", bb)
    hpBg.BackgroundColor3 = Color3.new(0, 0, 0)
    hpBg.BorderSizePixel = 0
    hpBg.Size = UDim2.new(0, 3, 1, 0)
    hpBg.Position = UDim2.new(0, -6, 0, 0)
    local hpFill = Instance.new("Frame", hpBg)
    hpFill.BackgroundColor3 = Color3.fromRGB(0, 255, 0)
    hpFill.BorderSizePixel = 0
    hpFill.Size = UDim2.new(1, 0, 1, 0)

    espCache[p] = { bb = bb, box = box, stroke = stroke, name = name, hpBg = hpBg, hpFill = hpFill }
    bb.Parent = CoreGui
end

local function removeESP(p)
    local d = espCache[p]
    if d then d.bb:Destroy() end
    espCache[p] = nil
end

Players.PlayerAdded:Connect(function(p) if p ~= LP then makeESP(p) end end)
Players.PlayerRemoving:Connect(removeESP)
for _, p in ipairs(Players:GetPlayers()) do if p ~= LP then makeESP(p) end end

RunService.RenderStepped:Connect(function()
    for p, d in pairs(espCache) do
        local char = p.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        local root = char and char:FindFirstChild("HumanoidRootPart")
        if not (Config.ESP.Enabled and hum and hum.Health > 0 and root) then
            d.bb.Enabled = false
            continue
        end
        local dist = (Camera.CFrame.Position - root.Position).Magnitude
        if dist > Config.ESP.MaxDistance then
            d.bb.Enabled = false
            continue
        end
        d.bb.Adornee = root
        d.bb.Enabled = true

        local team = Config.ESP.TeamCheck and isTeammate(p)
        local col = team and Color3.fromRGB(255, 80, 80) or Config.ESP.Color
        d.stroke.Color = col
        d.name.TextColor3 = col
        d.name.Text = string.format("%s [%d]", p.Name, math.floor(dist))
        d.name.Visible = Config.ESP.Name
        d.box.Visible = Config.ESP.Box
        d.stroke.Enabled = Config.ESP.Box

        if Config.ESP.Health then
            local hp = math.clamp(hum.Health / hum.MaxHealth, 0, 1)
            d.hpBg.Visible = true
            d.hpFill.Size = UDim2.new(1, 0, hp, 0)
            d.hpFill.Position = UDim2.new(0, 0, 1 - hp, 0)
            d.hpFill.BackgroundColor3 = Color3.fromRGB(255 * (1-hp), 255 * hp, 0)
        else
            d.hpBg.Visible = false
        end
    end
end)
