--==============================================================
-- Connection Registry
--==============================================================

local Connections = {}

local function connect(signal, callback)
    local connection = signal:Connect(callback)
    table.insert(Connections, connection)
    return connection
end

local function disconnectAll()
    for _, connection in ipairs(Connections) do
        if connection and connection.Connected then
            connection:Disconnect()
        end
    end
    table.clear(Connections)
end

--==============================================================
-- Character Helpers
--==============================================================

local function humanoidOf(model)
    return model and model:FindFirstChildOfClass("Humanoid")
end

local function rootOf(model)
    if not model or not model:IsA("Model") then
        return nil
    end
    return model:FindFirstChild("HumanoidRootPart") or model.PrimaryPart
end

local function validCharacter(model)
    return model
        and model:IsA("Model")
        and model ~= LocalPlayer.Character
        and rootOf(model) ~= nil
        and humanoidOf(model) ~= nil
end

--==============================================================
-- Target Service
--==============================================================

local TargetService = {
    Items = {},
    Labels = {},
    SelectedLabel = nil,
    SelectedModel = nil,
    TrackedPlayer = nil,
}

function TargetService:Clear()
    table.clear(self.Items)
    table.clear(self.Labels)
end

function TargetService:Add(label, model)
    if not validCharacter(model) then
        return
    end
    if self.Items[label] then
        return
    end
    self.Items[label] = model
    table.insert(self.Labels, label)
end

function TargetService:Refresh()
    self:Clear()

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            local label = string.format("PLAYER  •  %s  (@%s)", player.DisplayName, player.Name)
            self:Add(label, player.Character)
        end
    end

    for _, object in ipairs(Workspace:GetDescendants()) do
        if object:IsA("Model") and not Players:GetPlayerFromCharacter(object) then
            if validCharacter(object) then
                self:Add("NPC  •  " .. object.Name, object)
            end
        end
    end

    table.sort(self.Labels)
    return self.Labels
end

function TargetService:Select(label)
    local model = self.Items[label]
    if not validCharacter(model) then
        self.SelectedLabel = nil
        self.SelectedModel = nil
        return false
    end

    self.SelectedLabel = label
    self.SelectedModel = model

    local player = Players:GetPlayerFromCharacter(model)
    if player then
        self:TrackPlayer(player)
    else
        self:StopTracking()
    end

    return true
end

function TargetService:Get()
    if self.SelectedModel and self.SelectedModel.Parent and rootOf(self.SelectedModel) then
        return self.SelectedModel
    end
    return nil
end

function TargetService:StopTracking()
    self.TrackedPlayer = nil
end

function TargetService:TrackPlayer(player)
    self.TrackedPlayer = player
end

--==============================================================
-- Local Defense Service
--==============================================================

local DefenseService = {
    Active = false,
    Saved = {},
}

function DefenseService:Save(part)
    if self.Saved[part] then
        return
    end
    self.Saved[part] = {
        CanCollide = part.CanCollide,
        CanTouch = part.CanTouch,
        CanQuery = part.CanQuery,
    }
end

function DefenseService:Enable()
    self.Active = true
    local character = LocalPlayer.Character
    if not character then return end

    for _, object in ipairs(character:GetDescendants()) do
        if object:IsA("BasePart") then
            self:Save(object)
            object.CanTouch = false
            object.CanQuery = false
            if Config.NoCollision then
                object.CanCollide = false
            end
        end
    end
end

function DefenseService:Step()
    if not self.Active then return end
    local character = LocalPlayer.Character
    if not character then return end

    for _, object in ipairs(character:GetDescendants()) do
        if object:IsA("BasePart") then
            self:Save(object)
            object.CanTouch = false
            object.CanQuery = false
            if Config.NoCollision then
                object.CanCollide = false
            end
        end
    end

    if Config.ZeroVelocity then
        local root = rootOf(character)
        if root then
            root.AssemblyLinearVelocity = Vector3.zero
            root.AssemblyAngularVelocity = Vector3.zero
        end
    end
end

function DefenseService:Disable()
    self.Active = false
    for part, values in pairs(self.Saved) do
        if part and part.Parent then
            part.CanCollide = values.CanCollide
            part.CanTouch = values.CanTouch
            part.CanQuery = values.CanQuery
        end
    end
    table.clear(self.Saved)
end

--==============================================================
-- Mount Service
--==============================================================

local MountService = {
    Enabled = false,
}

function MountService:Calculate(target)
    local targetRoot = rootOf(target)
    if not targetRoot then return nil end

    local position = targetRoot.Position
        - targetRoot.CFrame.LookVector * Config.Distance
        + Vector3.new(0, Config.Height, 0)

    return CFrame.lookAt(
        position,
        position + targetRoot.CFrame.LookVector
    )
end

function MountService:Step()
    if not self.Enabled then return end

    local character = LocalPlayer.Character
    local target = TargetService:Get()
    if not character or not target then return end

    local humanoid = humanoidOf(character)
    if not humanoid or humanoid.Health <= 0 then return end

    local targetHumanoid = humanoidOf(target)
    if targetHumanoid and targetHumanoid.Health <= 0 then
        return
    end

    local destination = self:Calculate(target)
    if destination then
        character:PivotTo(destination)
        DefenseService:Step()
    end
end

function MountService:Enable()
    if not TargetService:Get() then
        return false
    end
    self.Enabled = true
    DefenseService:Enable()
    return true
end

function MountService:Disable()
    self.Enabled = false
    if not Config.SafeMode and not Config.AutoShield then
        DefenseService:Disable()
    end
end

    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        windowStart = Window.Position
    end

connect(UserInputService.InputChanged, function(input)
    if not dragging then return end
    if input.UserInputType ~= Enum.UserInputType.MouseMovement
        and input.UserInputType ~= Enum.UserInputType.Touch then
        return
    end

    local delta = input.Position - dragStart
    Window.Position = UDim2.new(
        windowStart.X.Scale,
        windowStart.X.Offset + delta.X,
        windowStart.Y.Scale,
        windowStart.Y.Offset + delta.Y
    )
end)

connect(UserInputService.InputEnded, function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
        dragging = false
    end
end)

    holder.BackgroundColor3 = Color3.fromRGB(27,27,36)

    text.BackgroundTransparency = 1
    text.TextColor3 = Color3.fromRGB(215,215,225)
    text.Font = Enum.Font.GothamMedium
    text.TextSize = 10
    text.TextXAlignment = Enum.TextXAlignment.Left
    text.Parent = holder

    local bar = Instance.new("TextButton")
    bar.Size = UDim2.new(1,-18,0,7)
    bar.Position = UDim2.fromOffset(9,34)
    bar.BackgroundColor3 = Color3.fromRGB(55,55,68)
    bar.Text = ""
    bar.AutoButtonColor = false
    bar.Parent = holder

    local fill = Instance.new("Frame")
    fill.BorderSizePixel = 0
    fill.BackgroundColor3 = Color3.fromRGB(115,115,245)
    fill.Parent = bar

    local fc = Instance.new("UICorner")
    fc.CornerRadius = UDim.new(1,0)
    fc.Parent = fill

    local valueNow = value
    local draggingSlider = false

    local function set(v)
        valueNow = math.clamp(v,min,max)
        local alpha = (valueNow-min)/(max-min)
        fill.Size = UDim2.new(alpha,0,1,0)
        text.Text = string.format("%s  %.1f",label,valueNow)
        callback(valueNow)
    end

    local function fromX(x)
        local alpha = math.clamp((x-bar.AbsolutePosition.X)/bar.AbsoluteSize.X,0,1)
        set(min+(max-min)*alpha)
    end

    connect(bar.MouseButton1Down,function(x)
        draggingSlider=true
        fromX(x)
    end)

    connect(UserInputService.InputChanged,function(input)
        if not draggingSlider then return end
        if input.UserInputType == Enum.UserInputType.MouseMovement then
            fromX(input.Position.X)
        end
    end)

    connect(UserInputService.InputEnded,function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            draggingSlider=false
        end
    end)

    set(value)
end

--==============================================================
-- Runtime Simulation
--==============================================================

connect(RunService.PreSimulation,function()
    if MountService.Enabled then
        MountService:Step()
    end

    if Config.SafeMode then
        DefenseService:Enable()
    end

    if Config.AutoShield then
        local localRoot = rootOf(LocalPlayer.Character)
        local target = TargetService:Get()
        local targetRoot = rootOf(target)

        if localRoot and targetRoot then
            if (localRoot.Position-targetRoot.Position).Magnitude <= Config.ShieldRadius then
                DefenseService:Enable()
                DefenseService:Step()
            end
        end
    end

    if Config.HealthLock then
        local humanoid = humanoidOf(LocalPlayer.Character)
        if humanoid and humanoid.Health > 0 then
            humanoid.Health = humanoid.MaxHealth
        end
    end
end)

        TargetService:Refresh()
    end)

connect(Players.PlayerRemoving,function(player)
    if TargetService.TrackedPlayer == player then
        if Config.AutoRemount then
            MountService:Disable()
        end
        TargetService:StopTracking()
    end
    task.defer(function()
        TargetService:Refresh()
    end)
end)

connect(Workspace.DescendantAdded,function(object)
    if object:IsA("Model") then
        task.defer(function()
            TargetService:Refresh()
        end)
    end
end)

connect(Workspace.DescendantRemoving,function(object)
    if object == TargetService.SelectedModel then
        if Config.AutoRemount then
            local player = Players:GetPlayerFromCharacter(object)
            if player then
                TargetService:TrackPlayer(player)
            end
        end
    end
end)

connect(Players.PlayerAdded,function(player)
    connect(player.CharacterAdded,function(character)
        if player == TargetService.TrackedPlayer and Config.AutoRemount then
            task.wait(0.35)
            TargetService.SelectedModel = character
            if MountService.Enabled then
                DefenseService:Enable()
            end
        end
    end)
end)

-- Existing players need respawn listeners too.
for _, player in ipairs(Players:GetPlayers()) do
    if player ~= LocalPlayer then
        connect(player.CharacterAdded,function(character)
            if player == TargetService.TrackedPlayer and Config.AutoRemount then
                task.wait(0.35)
                TargetService.SelectedModel = character
                if MountService.Enabled then
                    DefenseService:Enable()
                end
            end
        end)
    end
end

connect(LocalPlayer.CharacterAdded,function()
    task.wait(0.35)
    if MountService.Enabled or Config.SafeMode or Config.AutoShield then
        DefenseService:Enable()

end)
