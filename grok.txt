-- Prevent game from enumerating our hooks via debug.info.
pcall(function()
    local oldInfo = debug.info
    debug.info = function(...)
        local ok, res = pcall(oldInfo, ...)
        if ok and type(res) == "table" and res.source then
            if string.find(tostring(res.source), "Lunar") then
                res.source = "CoreScript"
            end
        end
        return res
    end
end)

-- Hide our UI from game:GetGuiObjectsAtPosition and CoreGui scans by
-- placing it in a randomly-named folder inside PlayerGui with a spoofed
-- class name (some ACs iterate PlayerGui by name).
local function secureParent(instance)
    local ok = pcall(function() instance.Parent = CoreGui end)
    if not ok then
        local pg = LocalPlayer:FindFirstChildOfClass("PlayerGui")
            or LocalPlayer:WaitForChild("PlayerGui")
        local hidden = pg:FindFirstChild("RobloxChat") or Instance.new("Folder", pg)
        hidden.Name = "RobloxChat"
        instance.Parent = hidden
    end
end

-- ============ HELPERS ============
local function isAlive(plr)
    local char = plr.Character
    if not char then return false end
    local hum = char:FindFirstChildOfClass("Humanoid")
    return hum and hum.Health > 0
end

local function getTargetPart(plr, mode)
    local char = plr.Character
    if not char then return nil end
    if mode == "Head" then
        return char:FindFirstChild("Head")
    else
        return char:FindFirstChild("UpperTorso")
            or char:FindFirstChild("Torso")
            or char:FindFirstChild("HumanoidRootPart")
    end
end

local function isVisible(part, ignoreList)
    if not part then return false end
    local origin = Camera.CFrame.Position
    local dir = part.Position - origin
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = ignoreList or { LocalPlayer.Character, Camera }
    params.IgnoreWater = true
    local result = Workspace:Raycast(origin, dir, params)
    return result == nil or result.Instance:IsDescendantOf(part.Parent)
end

local function predict(part, velocity, amount)
    if not part or not velocity then return part.Position end
    return part.Position + (velocity * (amount or 0.16))
end

-- ============ TARGET SELECTION ============
local function getClosestPlayer(fov, teamCheck, mode, wallCheck)
    local closest, shortest = nil, fov
    local center = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)

    for _, plr in ipairs(Players:GetPlayers()) do
        if plr == LocalPlayer then continue end
        if not isAlive(plr) then continue end
        if teamCheck and plr.Team == LocalPlayer.Team then continue end

        local part = getTargetPart(plr, mode)
        if not part then continue end

        local screenPos, onScreen = Camera:WorldToViewportPoint(part.Position)
        if not onScreen then continue end

        local dist = (Vector2.new(screenPos.X, screenPos.Y) - center).Magnitude
        if dist < shortest then
            if wallCheck and not isVisible(part, { LocalPlayer.Character, Camera }) then
                continue
            end
            shortest = dist
            closest = plr
        end
    end
    return closest
end

RunService.RenderStepped:Connect(function()
    if not Config.Aimbot.Enabled or not aimbotHeld then return end
    local target = getClosestPlayer(Config.Aimbot.FOV, Config.Aimbot.TeamCheck, Config.Aimbot.Mode, Config.Aimbot.WallCheck)
    if not target then return end

    local part = getTargetPart(target, Config.Aimbot.Mode)
    local root = target.Character and target.Character:FindFirstChild("HumanoidRootPart")
    if not part then return end

    local aimPos = part.Position
    if root and Config.Aimbot.Prediction > 0 then
        aimPos = predict(part, root.AssemblyLinearVelocity, Config.Aimbot.Prediction)
    end

    local desired = CFrame.new(Camera.CFrame.Position, aimPos)
    Camera.CFrame = Camera.CFrame:Lerp(desired, 1 - Config.Aimbot.Smoothness)
end)

-- ============ SILENT AIM ============
-- Hooks both legacy FindPartOnRay* and modern Raycast APIs.
local mt = getrawmetatable(game)
local oldNamecall = mt.__namecall
setreadonly(mt, false)
mt.__namecall = newcclosure(function(self, ...)
    local method = getnamecallmethod()
    local args = {...}

    if Config.SilentAim.Enabled and not checkcaller() then
        -- Random hit-chance gate
        if math.random(1, 100) > Config.SilentAim.HitChance then
            return oldNamecall(self, ...)
        end

        local target = getClosestPlayer(
            Config.SilentAim.FOV, Config.SilentAim.TeamCheck,
            Config.SilentAim.Mode, Config.SilentAim.WallCheck
        )
        if target then
            local part = getTargetPart(target, Config.SilentAim.Mode)
            local root = target.Character and target.Character:FindFirstChild("HumanoidRootPart")
            if part then
                local aimPos = part.Position
                if root and Config.SilentAim.Prediction > 0 then
                    aimPos = predict(part, root.AssemblyLinearVelocity, Config.SilentAim.Prediction)
                end

                -- Legacy ray methods
                if method == "FindPartOnRay" or method == "FindPartOnRayWithIgnoreList"
                   or method == "FindPartOnRayWithWhitelist" then
                    local oldRay = args[1]
                    if typeof(oldRay) == "Ray" then
                        args[1] = Ray.new(oldRay.Origin, aimPos - oldRay.Origin)
                    end
                    return oldNamecall(self, unpack(args))
                end

                -- Modern Raycast
                if method == "Raycast" and typeof(self) == "Instance" and self:IsA("Workspace") then
                    local origin = args[1]
                    local params = args[3]
                    if typeof(origin) == "Vector3" then
                        args[2] = (aimPos - origin).Unit * (args[2] and args[2].Magnitude or (aimPos - origin).Magnitude)
                    end
                    return oldNamecall(self, unpack(args))
                end
            end
        end
    end
    return oldNamecall(self, ...)
end)
setreadonly(mt, true)

    healthFill.BorderSizePixel = 0
    healthFill.Size = UDim2.new(1, 0, 1, 0)

local function removeGui(plr)
    local data = espCache[plr]
    if not data then return end
    if data.billboard then data.billboard:Destroy() end
    espCache[plr] = nil
end

local SKELETON = {
    {"Head","UpperTorso"},{"UpperTorso","LowerTorso"},
    {"UpperTorso","LeftUpperArm"},{"LeftUpperArm","LeftLowerArm"},{"LeftLowerArm","LeftHand"},
    {"UpperTorso","RightUpperArm"},{"RightUpperArm","RightLowerArm"},{"RightLowerArm","RightHand"},
    {"LowerTorso","LeftUpperLeg"},{"LeftUpperLeg","LeftLowerLeg"},{"LeftLowerLeg","LeftFoot"},
    {"LowerTorso","RightUpperLeg"},{"RightUpperLeg","RightLowerLeg"},{"RightLowerLeg","RightFoot"},
}

RunService.RenderStepped:Connect(function()
    for plr, data in pairs(espCache) do
        local char = plr.Character
        local hum  = char and char:FindFirstChildOfClass("Humanoid")
        local root = char and char:FindFirstChild("HumanoidRootPart")

        if not (Config.ESP.Enabled and char and hum and hum.Health > 0 and root) then
            data.billboard.Enabled = false
            continue
        end

        local dist = (Camera.CFrame.Position - root.Position).Magnitude
        if dist > Config.ESP.MaxDistance then
            data.billboard.Enabled = false
            continue
        end

        data.billboard.Adornee = root
        data.billboard.Enabled = true

        local team = Config.ESP.TeamCheck and plr.Team == LocalPlayer.Team
        local color = team and Config.ESP.TeamColor or Config.ESP.Color
        data.boxStroke.Color = color
        data.nameLabel.TextColor3 = color
        data.distLabel.TextColor3 = color

        data.nameLabel.Text = plr.Name
        data.nameLabel.Visible = Config.ESP.Name
        data.box.Visible = Config.ESP.Box
        data.boxStroke.Enabled = Config.ESP.Box

        -- Distance label
        if Config.ESP.Distance then
            data.distLabel.Text = string.format("[%d]", math.floor(dist))
            data.distLabel.Visible = true
        else
            data.distLabel.Visible = false
        end

        -- Health
        if Config.ESP.Health then
            local hp = math.clamp(hum.Health / hum.MaxHealth, 0, 1)
            data.healthBg.Visible = true
            data.healthFill.Size = UDim2.new(1, 0, hp, 0)
            data.healthFill.Position = UDim2.new(0, 0, 1 - hp, 0)
            data.healthFill.BackgroundColor3 = Color3.fromRGB(255 * (1 - hp), 255 * hp, 0)
        else
            data.healthBg.Visible = false
        end

        -- Skeleton
        if Config.ESP.Skeleton then
            for i, pair in ipairs(SKELETON) do
                local a = char:FindFirstChild(pair[1])
                local b = char:FindFirstChild(pair[2])
                local line = data.lines[i]
                if a and b then
                    local aPos, aOn = Camera:WorldToViewportPoint(a.Position)
                    local bPos, bOn = Camera:WorldToViewportPoint(b.Position)
                    -- Convert to billboard-local space: use screen delta directly
                    if aOn and bOn then
                        local delta = Vector2.new(bPos.X - aPos.X, bPos.Y - aPos.Y)
                        local length = delta.Magnitude
                        local angle = math.deg(math.atan2(delta.Y, delta.X))
                        line.Size = UDim2.new(0, length, 0, 1)
                        line.Position = UDim2.new(0, aPos.X, 0, aPos.Y)  -- screen space
                        line.Rotation = angle
                        line.Visible = true
                        line.BackgroundColor3 = color
                    else
                        line.Visible = false
                    end
                else
                    line.Visible = false
                end
            end
        else
            for _, line in pairs(data.lines) do line.Visible = false end
        end
    end
end)

    btn.Font = Enum.Font.Gotham

    btn.BorderSizePixel = 0
    btn.MouseButton1Click:Connect(function()
        local cur = getter()

        for i, v in ipairs(options) do if v == cur then idx = i end end
        idx = idx % #options + 1
        setter(options[idx])
        btn.Text = "> " .. name .. ": " .. getter()
    end)
    y = y + 28
end
