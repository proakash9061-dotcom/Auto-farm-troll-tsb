-- Make Target_A slowly move + rotate so you can see the tape effect
local moveConn
local function startTargetMovement(targetModel)
	if moveConn then moveConn:Disconnect() end
	local hrp = targetModel.PrimaryPart
	local t = 0
	moveConn = RunService.Heartbeat:Connect(function(dt)
		t += dt
		hrp.CFrame = CFrame.new(
			math.sin(t * 0.6) * 12,
			5 + math.sin(t * 1.2) * 1.5,
			math.cos(t * 0.45) * 10
		) * CFrame.Angles(0, t * 0.8, 0)
	end)
end
startTargetMovement(targets[1])

		if child:IsA("TextButton") then child:Destroy() end
	end
	for i, t in ipairs(targets) do
		local btn = Instance.new("TextButton")
		btn.Size = UDim2.new(1, -8, 0, 26)
		btn.BackgroundColor3 = (t == selectedTarget) and Color3.fromRGB(50, 90, 140) or Color3.fromRGB(35, 35, 42)
		btn.Text = "  " .. t.Name
		btn.TextColor3 = Color3.fromRGB(230, 230, 240)
		btn.Font = Enum.Font.Gotham
		btn.TextSize = 13
		btn.TextXAlignment = Enum.TextXAlignment.Left
		btn.Parent = targetList
		Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 4)
		
		btn.MouseButton1Click:Connect(function()
			selectedTarget = t
			startTargetMovement(t)
			refreshList()
		end)
	end
	targetList.CanvasSize = UDim2.new(0, 0, 0, #targets * 30)
end

-- ========== TAPE METHOD LOGIC ==========
local lastTargetPos = Vector3.zero
local lastTargetTime = tick()

RunService.RenderStepped:Connect(function()
	if not selectedTarget or not selectedTarget.PrimaryPart then return end
	
	local targetCF = selectedTarget.PrimaryPart.CFrame
	local desiredCF = targetCF * offset   -- THE TAPE METHOD
	
	-- Move player dummy + marker
	playerDummy.PrimaryPart.CFrame = desiredCF
	playerMarker.CFrame = desiredCF
	
	-- Target marker
	targetMarker.CFrame = targetCF
	
	-- Tape line (visual)
	local mid = (desiredCF.Position + targetCF.Position) / 2
	local dist = (desiredCF.Position - targetCF.Position).Magnitude
	tapeLine.Size = Vector3.new(0.12, 0.12, dist)
	tapeLine.CFrame = CFrame.lookAt(mid, targetCF.Position) * CFrame.new(0, 0, 0)
	
	-- Prediction (simple velocity based)
	if predictionOn then
		local now = tick()
		local dt = now - lastTargetTime
		if dt > 0.001 then
			local velocity = (targetCF.Position - lastTargetPos) / dt
			local futurePos = targetCF.Position + velocity * 0.35 -- look ahead
			local futureCF = CFrame.new(futurePos) * (targetCF - targetCF.Position)
			predictionMarker.CFrame = futureCF * offset
		end
		lastTargetPos = targetCF.Position
		lastTargetTime = now
	end
end)
