--[[
    Universal Mount & Attach System
    Roblox Studio / Private Experience Edition
    ------------------------------------------------------------
    Purpose:
      Educational character-positioning utility for experiences
      you own or are authorized to test.

    Features:
      * Vertical Rayfield-style UI (self-contained fallback UI)
      * Player + NPC/Dummy target discovery
      * DisplayName + Username formatting
      * Mount distance: 0.5 -> 5 studs
      * Height offset: -3 -> +3 studs
      * RunService.PreSimulation positioning
      * Model:PivotTo() attachment
      * Collision / touch / query protection
      * Linear + angular velocity protection
      * Dynamic target refresh
      * Connection cleanup
      * Respawn-safe character handling
      * No-keybind minimize / restore / destroy controls
      * Safe Mode + proximity Auto Shield
      * Optional local invulnerability test mode
      * Automatic re-attachment after player respawn

    IMPORTANT:
      This source is intended for Roblox Studio/private experiences.
      If you use Rayfield, replace the UI adapter section with your
      authorized Rayfield build if desired. This file intentionally
      contains a self-contained GUI so it can be tested without a
      third-party loader.
]]

--// Services
local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local Workspace = game:GetService("Workspace")

local LocalPlayer = Players.LocalPlayer

--// Configuration
local CONFIG = {
    MountDistance = 2,
    HeightOffset = 0,

    -- Set true to make the local character non-collidable while mounted.
    NoCollisionWhileMounted = true,

    -- Defensive physics handling for the local character.
    ZeroAssemblyVelocity = true,

    -- Only accept models containing a Humanoid and a root part.
    RequireHumanoid = true,
}

--==============================================================
-- Utility
--==============================================================

local function getCharacterRoot(model)
    if not model or not model:IsA("Model") then
        return nil
    end

    return model:FindFirstChild("HumanoidRootPart")
        or model.PrimaryPart
end

local function getHumanoid(model)
    if not model or not model:IsA("Model") then
        return nil
    end

    return model:FindFirstChildOfClass("Humanoid")
end

local function isValidTarget(model)
    if not model or not model:IsA("Model") then
        return false
    end

    if model == LocalPlayer.Character then
        return false
    end

    local root = getCharacterRoot(model)

    if not root then
        return false
    end

    if CONFIG.RequireHumanoid and not getHumanoid(model) then
        return false
    end

    return true
end

local function formatPlayerTarget(player)
    return string.format(
        "PLAYER  |  %s  (@%s)",
        player.DisplayName,
        player.Name
    )
end

local function formatNpcTarget(model)
    return string.format("NPC  |  %s", model.Name)
end

--==============================================================
-- Target Manager
--==============================================================

local TargetManager = {}
TargetManager.Targets = {}
TargetManager.Selected = nil

function TargetManager:Clear()
    table.clear(self.Targets)
end

function TargetManager:Add(label, model)
    if isValidTarget(model) then
        self.Targets[label] = model
    end
end

function TargetManager:Refresh()
    self:Clear()

    -- Players
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            if isValidTarget(player.Character) then
                self:Add(formatPlayerTarget(player), player.Character)
            end
        end
    end

    -- Workspace NPCs / Dummies
    for _, instance in ipairs(Workspace:GetDescendants()) do
        if instance:IsA("Model")
            and instance ~= LocalPlayer.Character
            and not Players:GetPlayerFromCharacter(instance)
            and isValidTarget(instance) then

            self:Add(formatNpcTarget(instance), instance)
        end
    end

    return self:GetLabels()
end

function TargetManager:GetLabels()
    local labels = {}

    for label in pairs(self.Targets) do
        table.insert(labels, label)
    end

    table.sort(labels)

    return labels
end

function TargetManager:SetSelected(label)
    local target = self.Targets[label]

    if target and isValidTarget(target) then
        self.Selected = target
        return true
    end

    self.Selected = nil
    return false
end

function TargetManager:GetSelected()
    if self.Selected and isValidTarget(self.Selected) then
        return self.Selected
    end

    self.Selected = nil
    return nil
end

--==============================================================
-- Character Protection
--==============================================================

local DefenseController = {}

DefenseController.Saved = {}
DefenseController.Active = false

function DefenseController:SavePart(part)
    if self.Saved[part] then
        return
    end

    self.Saved[part] = {
        CanCollide = part.CanCollide,
        CanTouch = part.CanTouch,
        CanQuery = part.CanQuery,
    }
end

function DefenseController:Enable()
    local character = LocalPlayer.Character

    if not character then
        return
    end

    self.Active = true

    for _, instance in ipairs(character:GetDescendants()) do
        if instance:IsA("BasePart") then
            self:SavePart(instance)

            if CONFIG.NoCollisionWhileMounted then
                instance.CanCollide = false
            end

            instance.CanTouch = false
            instance.CanQuery = false
        end
    end
end

function DefenseController:Apply()
    if not self.Active then
        return
    end

    local character = LocalPlayer.Character

    if not character then
        return
    end

    for _, instance in ipairs(character:GetDescendants()) do
        if instance:IsA("BasePart") then
            self:SavePart(instance)

            if CONFIG.NoCollisionWhileMounted then
                instance.CanCollide = false
            end

            instance.CanTouch = false
            instance.CanQuery = false
        end
    end
end

function DefenseController:Disable()
    self.Active = false

    for part, properties in pairs(self.Saved) do
        if part and part.Parent then
            part.CanCollide = properties.CanCollide
            part.CanTouch = properties.CanTouch
            part.CanQuery = properties.CanQuery
        end
    end

    table.clear(self.Saved)
end

function DefenseController:ZeroVelocity()
    if not self.Active or not CONFIG.ZeroAssemblyVelocity then
        return
    end

    local character = LocalPlayer.Character
    if not character then
        return
    end

    local root = getCharacterRoot(character)
    if not root then
        return
    end

    root.AssemblyLinearVelocity = Vector3.zero
    root.AssemblyAngularVelocity = Vector3.zero
end

--==============================================================
-- Mount Controller
--==============================================================

local MountController = {}

MountController.Enabled = false
MountController.Connection = nil
MountController.CharacterConnection = nil

function MountController:CalculateCFrame(target)
    local targetRoot = getCharacterRoot(target)

    if not targetRoot then
        return nil
    end

    -- Behind the target:
    -- target position - target LookVector * distance
    local behindPosition =
        targetRoot.Position
        - (targetRoot.CFrame.LookVector * CONFIG.MountDistance)

    local finalPosition =
        behindPosition
        + Vector3.new(0, CONFIG.HeightOffset, 0)

    -- Face the same direction as the target.
    return CFrame.lookAt(
        finalPosition,
        finalPosition + targetRoot.CFrame.LookVector
    )
end

function MountController:Step()
    if not self.Enabled then
        return
    end

    local character = LocalPlayer.Character
    local target = TargetManager:GetSelected()

    if not character or not target then
        return
    end

    local humanoid = getHumanoid(character)
    local root = getCharacterRoot(character)

    if not humanoid or not root then
        return
    end

    local targetRoot = getCharacterRoot(target)

    if not targetRoot then
        self:Disable()
        return
    end

    local mountCFrame = self:CalculateCFrame(target)

    if not mountCFrame then
        return
    end

    -- PivotTo is used instead of repeatedly setting individual parts.
    character:PivotTo(mountCFrame)

    DefenseController:Apply()
    DefenseController:ZeroVelocity()
end

function MountController:Enable()
    if self.Enabled then
        return
    end

    local target = TargetManager:GetSelected()

    if not target then
        return false, "Select a valid target first."
    end

    self.Enabled = true

    DefenseController:Enable()

    self.Connection = RunService.PreSimulation:Connect(function()
        self:Step()
    end)

    return true
end

function MountController:Disable()
    self.Enabled = false

    if self.Connection then
        self.Connection:Disconnect()
        self.Connection = nil
    end

    DefenseController:Disable()
end

function MountController:Toggle()
    if self.Enabled then
        self:Disable()
        return false
    end

    local success = self:Enable()

    return success == true
end

function MountController:Destroy()
    self:Disable()

    if self.CharacterConnection then
        self.CharacterConnection:Disconnect()
        self.CharacterConnection = nil
    end
end

MountController.CharacterConnection =
    LocalPlayer.CharacterAdded:Connect(function()
        task.wait(0.5)

        if MountController.Enabled then
            DefenseController:Enable()
        end
    end)

local function makeSlider(titleText, minValue, maxValue, initialValue, callback)
    local container = Instance.new("Frame")
    container.Size = UDim2.new(1, -8, 0, 60)
    container.BackgroundColor3 = Color3.fromRGB(25, 25, 33)
    container.Parent = scroll

    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(0, 8)
    c.Parent = container

    local label = Instance.new("TextLabel")
    label.Size = UDim2.new(1, -20, 0, 25)
    label.Position = UDim2.fromOffset(10, 4)
    label.BackgroundTransparency = 1
    label.TextColor3 = Color3.fromRGB(215, 215, 225)
    label.Font = Enum.Font.GothamMedium
    label.TextSize = 12
    label.TextXAlignment = Enum.TextXAlignment.Left
    label.Parent = container

    local bar = Instance.new("TextButton")
    bar.Size = UDim2.new(1, -20, 0, 8)
    bar.Position = UDim2.fromOffset(10, 38)
    bar.BackgroundColor3 = Color3.fromRGB(55, 55, 68)
    bar.Text = ""
    bar.AutoButtonColor = false
    bar.Parent = container

    local bc = Instance.new("UICorner")
    bc.CornerRadius = UDim.new(1, 0)
    bc.Parent = bar

    local fill = Instance.new("Frame")
    fill.BackgroundColor3 = Color3.fromRGB(110, 110, 255)
    fill.BorderSizePixel = 0
    fill.Parent = bar

    local fc = Instance.new("UICorner")
    fc.CornerRadius = UDim.new(1, 0)
    fc.Parent = fill

    local dragging = false
    local value = initialValue

    local function setValue(v)
        value = math.clamp(v, minValue, maxValue)

        local alpha = (value - minValue) / (maxValue - minValue)
        fill.Size = UDim2.new(alpha, 0, 1, 0)

        label.Text = string.format("%s: %.1f", titleText, value)

        callback(value)
    end

    local function fromX(x)
        local alpha =
            math.clamp(
                (x - bar.AbsolutePosition.X) / bar.AbsoluteSize.X,
                0,
                1
            )

        setValue(minValue + ((maxValue - minValue) * alpha))
    end

    bar.MouseButton1Down:Connect(function(x)
        dragging = true
        fromX(x)
    end)

    bar.MouseButton1Up:Connect(function()
        dragging = false
    end)

    bar.MouseMoved:Connect(function(x)
        if dragging then
            fromX(x)
        end
    end)

    setValue(initialValue)

    return {
        SetValue = setValue,
        GetValue = function()
            return value
        end,
    }
end

--==============================================================
-- Player / Workspace refresh events
--==============================================================

Players.PlayerAdded:Connect(function()
    task.defer(function()
        refreshTargets()
    end)
end)

Players.PlayerRemoving:Connect(function(player)
    local selected = TargetManager:GetSelected()

    if selected and Players:GetPlayerFromCharacter(selected) == player then
        MountController:Disable()
        mountButton.Text = "●  Mount: OFF"
    end

    task.defer(function()
        refreshTargets()
    end)
end)

Workspace.DescendantAdded:Connect(function(instance)
    if instance:IsA("Model") then
        task.defer(function()
            refreshTargets()
        end)
    end
end)

Workspace.DescendantRemoving:Connect(function(instance)
    if instance:IsA("Model") then
        if TargetManager.Selected == instance then
            MountController:Disable()
            mountButton.Text = "●  Mount: OFF"
        end

        task.defer(function()
            refreshTargets()
        end)
    end
end)

-- Keep the local test health stable when explicitly enabled.
-- This is NOT server-authoritative invincibility.
local healthConnection = RunService.PreSimulation:Connect(function()
    if not V2.LocalInvulnerabilityTest then
        return
    end

    local character = LocalPlayer.Character
    local humanoid = character and getHumanoid(character)

    if humanoid and humanoid.Health > 0 then
        humanoid.Health = humanoid.MaxHealth
    end
end)

-- Automatically re-select the player's new Character after death.
local trackedPlayer = nil
local trackedCharacterConnection = nil

local function stopTrackingPlayer()
    if trackedCharacterConnection then
        trackedCharacterConnection:Disconnect()
        trackedCharacterConnection = nil
    end

    trackedPlayer = nil
end

local function beginTrackingPlayer(player)
    stopTrackingPlayer()

    trackedPlayer = player

    if not V2.AutoRemount then
        return
    end

    trackedCharacterConnection = player.CharacterAdded:Connect(function(newCharacter)
        task.wait(0.35)

        if V2.Closed or not V2.AutoRemount then
            return
        end

        -- Re-point the selected target at the new character.
        TargetManager.Selected = newCharacter

        if MountController.Enabled then
            DefenseController:Enable()
        end
    end)
end

-- Wrap selection so player targets can be tracked for respawns.
local originalSetSelected = TargetManager.SetSelected

function TargetManager:SetSelected(label)
    local result = originalSetSelected(self, label)

    if result then
        local selected = self.Selected
        local player = selected and Players:GetPlayerFromCharacter(selected)

        if player then
            beginTrackingPlayer(player)
        else
            stopTrackingPlayer()
        end
    end

    return result
end

-- Auto shield checks proximity locally. It does not intercept
-- server-side RemoteEvents or guarantee immunity from server systems.
local autoShieldConnection = RunService.PreSimulation:Connect(function()
    if V2.Closed then
        return
    end

    if not V2.AutoShield then
        return
    end

    local character = LocalPlayer.Character
    local localRoot = character and getCharacterRoot(character)

    if not localRoot then
        return
    end

    local target = TargetManager:GetSelected()

    if target then
        local targetRoot = getCharacterRoot(target)

        if targetRoot then
            local distance = (localRoot.Position - targetRoot.Position).Magnitude

            if distance <= V2.AutoShieldRadius then
                DefenseController:Enable()
                DefenseController:ZeroVelocity()
            end
        end
    end
end)

-- Improved mount step:
-- If the selected player's character dies, keep tracking the player and
-- automatically attach again when their next character spawns.
local baseMountStep = MountController.Step

function MountController:Step()
    if not self.Enabled then
        return
    end

    local selected = TargetManager:GetSelected()

    if not selected then
        return
    end

    local targetHumanoid = getHumanoid(selected)

    if targetHumanoid and targetHumanoid.Health <= 0 then
        local player = Players:GetPlayerFromCharacter(selected)

        if player and V2.AutoRemount then
            if trackedPlayer ~= player then
                beginTrackingPlayer(player)
            end

            DefenseController:Enable()
        end

        return
    end

    baseMountStep(self)
end

-- Clean up V2 connections when the GUI is destroyed.
local originalDestroy = MountController.Destroy

function MountController:Destroy()
    stopTrackingPlayer()

    if healthConnection then
        healthConnection:Disconnect()
        healthConnection = nil
    end

    if autoShieldConnection then
        autoShieldConnection:Disconnect()
        autoShieldConnection = nil
    end

    originalDestroy(self)
end
